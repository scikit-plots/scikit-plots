"""Validate the cexternals/_annoy source-ownership and consumer contract."""
from __future__ import annotations

import argparse
import ast
import json
import pathlib
import re
import sys
from collections import Counter
from typing import Iterable

CYTHON_EXTENSIONS = {".pyx", ".pxd", ".pxi", ".in"}
SOURCE_EXTENSIONS = {".py", ".pyx", ".pxd", ".pxi", ".in", ".h", ".hpp", ".cc", ".cpp", ".cu"}
HEADER_RE = re.compile(r'extern\s+from\s+"[^"]*cexternals/_annoy/src/([^"]+)"')
FORBIDDEN_METADATA_KEYS = {"command", "commands", "shell", "exec", "executable", "script"}


def find_repo_root(start: pathlib.Path | None = None) -> pathlib.Path:
    """Find the wide repository root without assuming the script lives under scikitplot/."""
    here = (start or pathlib.Path(__file__)).resolve()
    if here.is_file():
        here = here.parent
    for candidate in (here, *here.parents):
        if all((candidate / name).is_dir() for name in ("scikitplot", "maintenances", "skills")):
            return candidate
    raise RuntimeError("could not locate wide repository root containing scikitplot/, maintenances/, skills/")


def safe_path(repo: pathlib.Path, rel: str) -> pathlib.Path:
    """Resolve a manifest path and reject absolute/path-escape inputs."""
    candidate = pathlib.Path(rel)
    if candidate.is_absolute():
        raise ValueError(f"absolute path is forbidden in maintenance metadata: {rel!r}")
    resolved = (repo / candidate).resolve()
    try:
        resolved.relative_to(repo.resolve())
    except ValueError as exc:
        raise ValueError(f"path escapes repository root: {rel!r}") from exc
    return resolved


def _walk_metadata(value: object, trail: str = "$") -> list[str]:
    problems: list[str] = []
    if isinstance(value, dict):
        for key, child in value.items():
            if str(key).lower() in FORBIDDEN_METADATA_KEYS:
                problems.append(f"{trail}.{key}: executable metadata key is forbidden")
            problems.extend(_walk_metadata(child, f"{trail}.{key}"))
    elif isinstance(value, list):
        for index, child in enumerate(value):
            problems.extend(_walk_metadata(child, f"{trail}[{index}]"))
    return problems


def load_manifest(repo: pathlib.Path) -> tuple[dict, list[str]]:
    path = repo / "maintenances/cexternals/_annoy/MAINTENANCE.json"
    problems: list[str] = []
    try:
        manifest = json.loads(path.read_text())
    except (OSError, json.JSONDecodeError) as exc:
        return {}, [f"cannot read MAINTENANCE.json: {exc}"]
    if manifest.get("schema_version") != 2:
        problems.append("MAINTENANCE.json schema_version must be 2")
    if manifest.get("subsystem") != "cexternals/_annoy":
        problems.append("MAINTENANCE.json subsystem must be cexternals/_annoy")
    problems.extend(_walk_metadata(manifest))
    for key in ("runtime_root", "shared_source", "maintenance_root", "skill_root"):
        value = manifest.get(key)
        if not isinstance(value, str) or not value:
            problems.append(f"MAINTENANCE.json {key} must be a non-empty relative path")
            continue
        try:
            safe_path(repo, value)
        except ValueError as exc:
            problems.append(str(exc))
    return manifest, problems


def _iter_code_files(root: pathlib.Path, suffixes: set[str]) -> Iterable[pathlib.Path]:
    for path in sorted(root.rglob("*")):
        if not path.is_file() or path.suffix not in suffixes:
            continue
        if "__pycache__" in path.parts or "_backup" in path.parts:
            continue
        yield path


def runtime_inventory(repo: pathlib.Path, manifest: dict) -> dict:
    runtime = safe_path(repo, manifest["runtime_root"])
    source_files = list(_iter_code_files(runtime, SOURCE_EXTENSIONS))
    tests = [p for p in source_files if "tests" in p.relative_to(runtime).parts]
    production = [p for p in source_files if p not in tests]
    def loc(paths: Iterable[pathlib.Path]) -> int:
        return sum(len(p.read_text(errors="replace").splitlines()) for p in paths)
    return {
        "source_files": len(production),
        "source_loc": loc(production),
        "test_files": len(tests),
        "test_loc": loc(tests),
        "markdown_files": len(list(runtime.rglob("*.md"))),
    }


def collect_source_edges(repo: pathlib.Path) -> dict[str, Counter[str]]:
    """Find every top-level module that directly names _annoy source headers."""
    package = repo / "scikitplot"
    result: dict[str, Counter[str]] = {}
    for top in sorted(package.iterdir()):
        if not top.is_dir() or top.name.startswith(".") or top.name == "cexternals":
            continue
        counts: Counter[str] = Counter()
        for path in _iter_code_files(top, CYTHON_EXTENSIONS):
            if "tests" in path.relative_to(top).parts:
                continue
            for line in path.read_text(errors="replace").splitlines():
                match = HEADER_RE.search(line)
                if match:
                    counts[match.group(1)] += 1
        if counts:
            result[top.name] = counts
    return result


def _normalized_imports(path: pathlib.Path, package_name: str) -> list[tuple[str, str]]:
    """Return normalized (module, symbol) imports from Python AST, including relative imports."""
    try:
        tree = ast.parse(path.read_text(errors="replace"))
    except SyntaxError:
        return []
    imports: list[tuple[str, str]] = []
    for node in ast.walk(tree):
        if isinstance(node, ast.ImportFrom):
            if node.level:
                base = package_name.split(".")[:-node.level]
                module_parts = (node.module or "").split(".") if node.module else []
                module = ".".join([*base, *module_parts])
            else:
                module = node.module or ""
            for name in node.names:
                imports.append((module, name.name))
        elif isinstance(node, ast.Import):
            for alias in node.names:
                imports.append((alias.name, ""))
    return imports


def file_imports(repo: pathlib.Path, rel: str) -> list[tuple[str, str]]:
    path = safe_path(repo, rel)
    rel_parts = path.relative_to(repo).with_suffix("").parts
    if rel_parts[0] != "scikitplot":
        return []
    package_name = ".".join(rel_parts)
    return _normalized_imports(path, package_name)


def module_imports(repo: pathlib.Path, runtime_root: str) -> list[tuple[str, str, str, int]]:
    root = safe_path(repo, runtime_root)
    found: list[tuple[str, str, str, int]] = []
    for path in _iter_code_files(root, {".py"}):
        if "tests" in path.relative_to(root).parts:
            continue
        try:
            tree = ast.parse(path.read_text(errors="replace"))
        except SyntaxError:
            continue
        rel_parts = path.relative_to(repo).with_suffix("").parts
        package_name = ".".join(rel_parts)
        for node in ast.walk(tree):
            if isinstance(node, ast.ImportFrom):
                if node.level:
                    base = package_name.split(".")[:-node.level]
                    parts = (node.module or "").split(".") if node.module else []
                    module = ".".join([*base, *parts])
                else:
                    module = node.module or ""
                for name in node.names:
                    found.append((module, name.name, str(path.relative_to(repo)), node.lineno))
            elif isinstance(node, ast.Import):
                for alias in node.names:
                    found.append((alias.name, "", str(path.relative_to(repo)), node.lineno))
    return found


def check_source_headers(repo: pathlib.Path, manifest: dict) -> list[str]:
    problems: list[str] = []
    shared = safe_path(repo, manifest["shared_source"])
    if not shared.is_dir():
        return [f"shared source directory is missing: {manifest['shared_source']}"]
    for header in manifest.get("required_headers", []):
        candidate = shared / header
        if not candidate.is_file():
            problems.append(f"required shared header is missing: {header}")
    actual = collect_source_edges(repo)
    expected_edges = {
        edge["consumer"]: Counter(edge["headers"])
        for edge in manifest.get("consumer_edges", [])
        if edge.get("kind") == "direct-source"
    }
    if set(actual) != set(expected_edges):
        missing = sorted(set(expected_edges) - set(actual))
        extra = sorted(set(actual) - set(expected_edges))
        if missing:
            problems.append(f"declared direct-source consumers no longer reference _annoy headers: {missing}")
        if extra:
            problems.append(f"undeclared direct-source consumers reference _annoy headers: {extra}")
    for consumer in sorted(set(actual) & set(expected_edges)):
        if actual[consumer] != expected_edges[consumer]:
            problems.append(
                f"{consumer}: shared-header references changed; expected {dict(expected_edges[consumer])}, "
                f"actual {dict(actual[consumer])}"
            )
    return problems


def check_compiled_consumers(repo: pathlib.Path, manifest: dict) -> list[str]:
    problems: list[str] = []
    for edge in manifest.get("consumer_edges", []):
        kind = edge.get("kind")
        if kind == "direct-source":
            continue
        evidence_file = edge.get("evidence_file")
        if not isinstance(evidence_file, str):
            problems.append(f"{edge.get('consumer')}: evidence_file missing")
            continue
        path = safe_path(repo, evidence_file)
        if not path.is_file():
            problems.append(f"{edge.get('consumer')}: evidence file missing: {evidence_file}")
            continue
        imports = set(file_imports(repo, evidence_file))
        for spec in edge.get("imports", []):
            module, _, symbol = spec.partition(":")
            if (module, symbol) not in imports:
                problems.append(f"{edge['consumer']}: required import evidence missing: {spec} in {evidence_file}")
        prefixes = tuple(edge.get("forbid_direct_import_prefixes", []))
        if prefixes:
            for module, symbol, rel, line in module_imports(repo, edge["runtime_root"]):
                if any(module == prefix or module.startswith(prefix + ".") for prefix in prefixes):
                    problems.append(
                        f"{edge['consumer']}: direct import {module}:{symbol or '*'} at {rel}:{line} "
                        "bypasses the declared transitive abstraction"
                    )
    return problems


def check_no_upward_imports(repo: pathlib.Path, manifest: dict) -> list[str]:
    problems: list[str] = []
    prefixes = tuple(manifest.get("forbidden_upward_import_prefixes", []))
    for module, symbol, rel, line in module_imports(repo, manifest["runtime_root"]):
        if any(module == p or module.startswith(p + ".") for p in prefixes):
            problems.append(f"runtime upward import forbidden: {module}:{symbol or '*'} at {rel}:{line}")
    return problems


def check_skill(repo: pathlib.Path, manifest: dict) -> list[str]:
    skill = safe_path(repo, manifest["skill_root"]) / "SKILL.md"
    if not skill.is_file():
        return ["fresh-chat skill is missing: skills/cexternals/_annoy/SKILL.md"]
    text = skill.read_text(errors="replace")
    required = ["MAINTENANCE.json", "FRESH_CHAT_HANDOFF.md", "direct-source", "compiled-index", "transitive-compiled-index"]
    return [f"SKILL.md is missing routing term {term!r}" for term in required if term not in text]


def check_residue(repo: pathlib.Path, manifest: dict) -> list[str]:
    root = safe_path(repo, manifest["runtime_root"])
    problems: list[str] = []
    for path in sorted(root.rglob("*")):
        if "__pycache__" in path.parts or path.suffix in {".pyc", ".pyo"}:
            problems.append(f"runtime residue must not ship: {path.relative_to(repo)}")
    return problems


def run_checks(repo: pathlib.Path) -> tuple[dict[str, list[str]], dict]:
    manifest, manifest_problems = load_manifest(repo)
    results: dict[str, list[str]] = {"manifest": manifest_problems}
    if manifest_problems:
        return results, manifest
    runtime = safe_path(repo, manifest["runtime_root"])
    if not runtime.is_dir():
        results["runtime_inventory"] = [f"runtime root missing: {manifest['runtime_root']}"]
    else:
        results["runtime_inventory"] = []
    results["source_headers"] = check_source_headers(repo, manifest)
    results["consumer_edges"] = check_compiled_consumers(repo, manifest)
    results["no_upward_imports"] = check_no_upward_imports(repo, manifest)
    results["fresh_chat_skill"] = check_skill(repo, manifest)
    results["residue"] = check_residue(repo, manifest)
    return results, manifest


def main(argv: list[str] | None = None) -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--repo", type=pathlib.Path, default=None, help="explicit repository root")
    parser.add_argument("--json", action="store_true", help="emit machine-readable result")
    parser.add_argument("--inventory", action="store_true", help="include current runtime inventory")
    args = parser.parse_args(argv)
    try:
        repo = (args.repo or find_repo_root()).resolve()
        results, manifest = run_checks(repo)
    except (RuntimeError, ValueError) as exc:
        print(f"ERROR: {exc}", file=sys.stderr)
        return 2
    ok = all(not items for items in results.values())
    payload = {"subsystem": "cexternals/_annoy", "status": "GREEN" if ok else "RED", "checks": results}
    if args.inventory and manifest:
        payload["inventory"] = runtime_inventory(repo, manifest)
    if args.json:
        print(json.dumps(payload, indent=2, sort_keys=True))
    else:
        for name, items in results.items():
            if items:
                for item in items:
                    print(f"RED {name}: {item}")
            else:
                print(f"GREEN {name}")
        if args.inventory and manifest:
            inv = runtime_inventory(repo, manifest)
            print("INVENTORY " + " ".join(f"{k}={v}" for k, v in inv.items()))
        print(f"cexternals/_annoy: {'GREEN' if ok else 'RED'}")
    return 0 if ok else 1


if __name__ == "__main__":
    raise SystemExit(main())
