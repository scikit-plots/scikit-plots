upload = widgets.FileUpload(accept=".csv", multiple=False)
display(upload)

def process_file(change=None):
    if not upload.value:
        return
    file_info = next(iter(upload.value))
    return pd.read_csv(io.BytesIO(file_info["content"]))

upload.observe(
  process_file,
  names="value",
)
